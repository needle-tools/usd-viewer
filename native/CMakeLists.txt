cmake_minimum_required(VERSION 3.24)
project(NeedleHydra LANGUAGES CXX OBJCXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_OBJCXX_STANDARD 17)
if(NOT APPLE)
 message(FATAL_ERROR "This experiment currently implements macOS IOSurface interop")
endif()
set(NEEDLE_HOST "usdview" CACHE STRING "usdview, blender or houdini")
set(NEEDLE_OPENUSD_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../openusd" CACHE PATH "USD source containing the shared Hydra instancer")
set(NEEDLE_ELECTRON "${CMAKE_CURRENT_SOURCE_DIR}/node_modules/electron/dist/Electron.app/Contents/MacOS/Electron" CACHE FILEPATH "Electron executable with linear HDR canvas and GPU shared-texture OSR")
if(NOT EXISTS "${NEEDLE_ELECTRON}")
 message(FATAL_ERROR "Set NEEDLE_ELECTRON to a shared-texture capable Electron executable")
endif()
add_library(hdNeedle SHARED src/plugin.mm src/transport.mm)
if(NEEDLE_HOST STREQUAL "blender")
 set(NEEDLE_BLENDER_APP "/Applications/Blender.app" CACHE PATH "Target Blender application")
 set(NEEDLE_BLENDER_SDK "${CMAKE_CURRENT_SOURCE_DIR}/.build/blender-sdk" CACHE PATH "Matching Blender libraries checkout")
 if(NOT EXISTS "${NEEDLE_BLENDER_SDK}/usd/include/pxr/pxr.h")
  message(FATAL_ERROR "A matching Blender USD SDK is required; do not use another host's headers")
 endif()
 target_include_directories(hdNeedle PRIVATE "${NEEDLE_BLENDER_SDK}/usd/include" "${NEEDLE_BLENDER_SDK}/opensubdiv/include" "${NEEDLE_BLENDER_SDK}/materialx/include" "${NEEDLE_BLENDER_SDK}/tbb/include" "${NEEDLE_BLENDER_APP}/Contents/Resources/5.1/python/include/python3.13" "${NEEDLE_BLENDER_SDK}/python/include/python3.13")
 target_link_libraries(hdNeedle PRIVATE "${NEEDLE_BLENDER_APP}/Contents/Resources/lib/libosdCPU.dylib")
 target_link_libraries(hdNeedle PRIVATE "${NEEDLE_BLENDER_APP}/Contents/Resources/lib/libusd_ms.dylib" "${NEEDLE_BLENDER_APP}/Contents/Resources/lib/libtbb.dylib" "${NEEDLE_BLENDER_APP}/Contents/Resources/lib/libMaterialXCore.dylib" "${NEEDLE_BLENDER_APP}/Contents/Resources/lib/libMaterialXFormat.dylib")
 target_link_options(hdNeedle PRIVATE -undefined dynamic_lookup)
 set_target_properties(hdNeedle PROPERTIES BUILD_RPATH "${NEEDLE_BLENDER_APP}/Contents/Resources/lib")
elseif(NEEDLE_HOST STREQUAL "houdini")
 set(NEEDLE_HOUDINI_ROOT "/Applications/Houdini/Current/Frameworks/Houdini.framework/Versions/Current" CACHE PATH "Target Houdini framework")
 target_include_directories(hdNeedle PRIVATE "${NEEDLE_HOUDINI_ROOT}/Resources/toolkit/include" "${NEEDLE_HOUDINI_ROOT}/../../../Python.framework/Versions/3.13/include/python3.13")
 foreach(library hd hdsi hdx hdMtlx hgi hgiMetal pxOsd usdImaging usdLux usdMtlx sdr arch gf js tf vt sdf hf python)
  target_link_libraries(hdNeedle PRIVATE "${NEEDLE_HOUDINI_ROOT}/Libraries/libpxr_${library}.dylib")
 endforeach()
 foreach(library MaterialXCore MaterialXFormat)
  target_link_libraries(hdNeedle PRIVATE "${NEEDLE_HOUDINI_ROOT}/Libraries/lib${library}.1.39.5.dylib")
 endforeach()
 target_link_libraries(hdNeedle PRIVATE "${NEEDLE_HOUDINI_ROOT}/Libraries/libosdCPU.dylib")
 target_link_libraries(hdNeedle PRIVATE "${NEEDLE_HOUDINI_ROOT}/../../../Python.framework/Versions/3.13/Python")
 set_target_properties(hdNeedle PROPERTIES BUILD_RPATH "${NEEDLE_HOUDINI_ROOT}/Libraries")
else()
 find_package(OpenGL REQUIRED)
 find_package(pxr CONFIG REQUIRED)
 target_include_directories(hdNeedle PRIVATE ${PXR_INCLUDE_DIRS})
 target_link_libraries(hdNeedle PRIVATE hd hdsi hdx hdMtlx hgi hgiMetal pxOsd osdCPU usdImaging usdLux usdMtlx sdr gf js tf vt sdf MaterialXCore MaterialXFormat)
endif()
target_include_directories(hdNeedle PRIVATE "${NEEDLE_OPENUSD_SOURCE}/pxr/usdImaging/hdEmscripten")
target_compile_definitions(hdNeedle PRIVATE NEEDLE_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}" NEEDLE_ELECTRON="${NEEDLE_ELECTRON}")
target_compile_definitions(hdNeedle PRIVATE NEEDLE_HOST_NAME="${NEEDLE_HOST}")
target_compile_options(hdNeedle PRIVATE -Wno-deprecated-declarations)
target_link_libraries(hdNeedle PRIVATE "-framework Metal" "-framework Foundation" "-framework IOSurface")
set_target_properties(hdNeedle PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugin")
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/plugin/resources")
configure_file(plugInfo.json "${CMAKE_BINARY_DIR}/plugin/resources/plugInfo.json" COPYONLY)
option(NEEDLE_BUILD_TESTS "Build focused native GPU interop diagnostics" OFF)
if(NEEDLE_BUILD_TESTS)
 add_executable(needle-gl-handoff-test tests/gl-handoff.mm experiments/gl-handoff/gl-handoff.mm)
 get_target_property(NEEDLE_TEST_INCLUDES hdNeedle INCLUDE_DIRECTORIES)
 get_target_property(NEEDLE_TEST_LIBRARIES hdNeedle LINK_LIBRARIES)
 target_include_directories(needle-gl-handoff-test PRIVATE src experiments/gl-handoff ${NEEDLE_TEST_INCLUDES})
 target_compile_options(needle-gl-handoff-test PRIVATE -Wno-deprecated-declarations)
 target_link_libraries(needle-gl-handoff-test PRIVATE hdNeedle ${NEEDLE_TEST_LIBRARIES} "-framework OpenGL")
endif()
